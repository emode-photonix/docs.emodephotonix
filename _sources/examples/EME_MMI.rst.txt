EME: MMI
--------

Two multimode interference (MMI) couplers in silicon at 1550 nm: a 1x2 splitter whose length is
swept around the self-imaging estimate, and a 2x2 coupler whose width and length are optimized
together for a 50/50 split with ``optimize()``.

Both couplers are silicon strips 220 nm thick in silica, built from three straight sections: the
access waveguides, the MMI body and the output waveguides. A window around each access waveguide
gives it its own port, so the S-matrix names each arm, and ``boundary_condition='TE-v'`` folds the
window to halve the solve.

The self-imaging lengths that follow from the beat length :math:`L_\pi` are only estimates for a
high-contrast guide, and the best images form a few percent shorter. A length sweep reuses the
solved modes and solves nothing, so the example sweeps the length first and lets the optimizer
refine from the best image. Light reflected between the two ends of the MMI body makes the split
ripple with the length, and each trough of that ripple is a local optimum, so the optimizer is kept
within one ripple period of its starting point.

The optimized 2x2 splits close to 50/50, with the through port leading the cross port by close to
the ideal 90 degrees, and it keeps a balanced split over several tens of nanometers around 1550 nm.
Every console report also says where the rest of the power goes: the higher-order modes of the
output waveguides, the reflection back into the input, and the light scattered into the cladding.
The access waveguides meet the MMI straight, with no tapers, which a fabricated device adds to lower
the loss.

        import sys
        import numpy as np
        import emodeconnection as emc
        from emodeconnection import Port, Bounds, Parameter, Objective
        from matplotlib import pyplot as plt

        ## Set simulation parameters
        wavelength = 1550  # [nm] wavelength
        dx, dy = 25, 25  # [nm] resolution
        h_core = 220  # [nm] silicon thickness
        w_access = 1000  # [nm] access waveguide width at the MMI
        pitch = 1500  # [nm] center-to-center spacing of the access waveguides
        access_length = 2000  # [nm] length of each access section
        w_1x2 = 3000  # [nm] 1x2 MMI width, its outputs at +/- w_1x2 / 4
        w_2x2 = 4500  # [nm] 2x2 MMI width, its ports at +/- w_2x2 / 6
        window_height = 2000  # [nm]
        num_modes = 30  # [-]

        ## The cross-sections are silicon strips in silica, symmetric top to bottom,
        ## so boundary_condition='TE-v' folds the window at its middle and keeps the
        ## TE-like modes, the ones an MMI for TE light is designed with. A window
        ## around each access waveguide gives it its own port.
        d = pitch / 2  # [nm] each access waveguide's distance from the center
        window = lambda x: Bounds(window_width=pitch, window_height=window_height, offset=(x, 0.0))
        arms = lambda side: [Port(name='a', side=side, window=window(-d)),
                             Port(name='b', side=side, window=window(d))]


        def mmi(name, width, num_inputs):
            """Declare an MMI with one or two inputs and two outputs, and return its beat length."""
            em = emc.EMode(emode_cmd=sys.argv[1:], simulation_name=name)
            em.settings(
                wavelength=wavelength, x_resolution=dx, y_resolution=dy,
                window_width=width + 2000, window_height=window_height,
                num_modes=num_modes, boundary_condition='TE-v',
                background_material='SiO2')
            em.shape(name='core', material='Si', height=h_core, etch_depth=h_core,
                     position=[0, window_height / 2], mask=w_access)
            em.label_profile(name='single')
            em.shape(name='core', mask=[w_access, w_access], mask_offset=[-d, d])
            em.label_profile(name='pair')

            ## The MMI body is the default profile, so a 'shape, core, mask' key
            ## changes its width. Its two lowest modes give the beat length L_pi.
            em.shape(name='core', mask=width, mask_offset=0)
            em.FDM()
            n_eff = np.real(np.asarray(em.get('effective_index')))
            beat_length = wavelength / (2 * (n_eff[0] - n_eff[1]))

            if num_inputs == 1:
                em.straight_section(name='in', profile='single', length=access_length)
            else:
                em.straight_section(name='in', profile='pair', length=access_length, ports=arms('left'))
            em.straight_section(name='mmi', profile='_default', length=beat_length)
            em.straight_section(name='out', profile='pair', length=access_length, ports=arms('right'))
            return em, beat_length


        def report(label, S, into):
            """Print where the power launched into port `into` goes."""
            a = np.abs(S.block('out.a', into)[0, 0]) ** 2
            b = np.abs(S.block('out.b', into)[0, 0]) ** 2
            reflected = np.sum(np.abs(S.block(into, into)[:, 0]) ** 2)
            higher = (np.sum(np.abs(S.block('out.a', into)[1:, 0]) ** 2)
                      + np.sum(np.abs(S.block('out.b', into)[1:, 0]) ** 2))
            print(f'{label} power in output a: {a:.4f}')
            print(f'{label} power in output b: {b:.4f}')
            print(f'{label} excess loss: {-10 * np.log10(a + b):.3f} dB')
            print(f'{label} imbalance, a over b: {round(10 * np.log10(a / b), 3) + 0.0:.3f} dB')
            print(f'{label} reflection: {10 * np.log10(reflected):.1f} dB')
            print(f'{label} power in higher-order output modes: {higher:.4f}')
            print(f'{label} power scattered out of the guides: {1 - a - b - reflected - higher:.4f}')


        ## ---------------------------------------------------------------- the 1x2
        ## One input on the axis forms two images at 3 L_pi / 8. That estimate uses
        ## only the two lowest modes, so the length is swept around it. A length
        ## sweep reuses the solved modes and solves nothing.
        em, beat_length = mmi('mmi_1x2', w_1x2, num_inputs=1)
        print(f'1x2 beat length L_pi: {beat_length:.0f} nm')
        print(f'1x2 self-imaging length 3 L_pi / 8: {3 * beat_length / 8:.0f} nm')

        lengths = 3 * beat_length / 8 * np.linspace(0.85, 1.1, 26)  # [nm]
        data = em.sweep(key='section, mmi, length', values=lengths, result='S_matrix')
        T = np.array([np.abs(S.block('out.a', 'in.left')[0, 0]) ** 2
                      + np.abs(S.block('out.b', 'in.left')[0, 0]) ** 2 for S in data['S_matrix']])
        best_length = lengths[np.argmax(T)]
        print(f'1x2 best length: {best_length:.0f} nm')

        em.straight_section(name='mmi', length=best_length)
        em.EME()
        report('1x2', em.get('S_matrix'), 'in.left')
        em.plot(component='Ex', plot_function='abs', plane='z-x', view='layout',
                file_name='mmi_1x2_field', file_type='png')
        em.close()

        plt.figure(figsize=(6, 3.5))
        plt.plot(lengths / 1e3, T)
        plt.axvline(3 * beat_length / 8e3, color='gray', ls='--', label='3 L$_\\pi$ / 8')
        plt.xlabel('MMI length (um)')
        plt.ylabel('Power in both outputs')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.savefig('mmi_1x2_length.png', dpi=300, bbox_inches='tight')

        ## ---------------------------------------------------------------- the 2x2
        ## Inputs at a third of the way from the axis to each edge excite only the
        ## modes that image in pairs, so a 50/50 split forms at L_pi / 2.
        em, beat_length = mmi('mmi_2x2', w_2x2, num_inputs=2)
        x0 = beat_length / 2
        print(f'2x2 beat length L_pi: {beat_length:.0f} nm')
        print(f'2x2 self-imaging length L_pi / 2: {x0:.0f} nm')

        ## How far both outputs are from 0.5, the quantity the optimizer minimizes
        through = Objective('S_power', out='out.a', into='in.a', goal='target', target=0.5)
        cross = Objective('S_power', out='out.b', into='in.a', goal='target', target=0.5)
        distance = lambda S: ((np.abs(S.block('out.a', 'in.a')[0, 0]) ** 2 - 0.5) ** 2
                              + (np.abs(S.block('out.b', 'in.a')[0, 0]) ** 2 - 0.5) ** 2)

        ## Light reflected at the two ends of the MMI body ripples the split with
        ## the length, with a period of about half a wavelength in the guide
        ## (270 nm), and every trough of that ripple is a local minimum. A length
        ## sweep at the starting width, which solves nothing, finds the best image.
        ## The optimizer then adjusts the width and the length together within
        ## 100 nm of it, less than half the ripple's period, so it stays in that
        ## trough.
        lengths = x0 * np.arange(0.9, 1.05, 0.002)  # [nm]
        data = em.sweep(key='section, mmi, length', values=lengths, result='S_matrix')
        start = lengths[np.argmin([distance(S) for S in data['S_matrix']])]
        print(f'2x2 best length at {w_2x2} nm wide: {start:.0f} nm')

        res = em.optimize(
            parameters=[Parameter('shape, core, mask', bounds=(w_2x2 - 300, w_2x2 + 300), x0=w_2x2),
                        Parameter('section, mmi, length', bounds=(start - 100, start + 100), x0=start)],
            objective=[through, cross],
            max_evaluations=80, tolerance=1e-5)
        width, length = res.x
        print(f'2x2 optimizer: {res.status} after {res.evaluations} evaluations in {res.seconds:.0f} s')
        print(f'2x2 optimized width: {width:.1f} nm')
        print(f'2x2 optimized length: {length:.0f} nm')

        ## The simulation is left solved at the best point
        S = em.get('S_matrix')
        report('2x2', S, 'in.a')
        phase_through = np.angle(S.block('out.a', 'in.a')[0, 0], deg=True)
        phase_cross = np.angle(S.block('out.b', 'in.a')[0, 0], deg=True)
        print(f'2x2 through-port phase: {phase_through:.1f} deg')
        print(f'2x2 cross-port phase: {phase_cross:.1f} deg')
        print(f'2x2 phase difference, through minus cross: '
              f'{(phase_through - phase_cross + 180) % 360 - 180:.1f} deg')
        em.plot(component='Ex', plot_function='abs', plane='z-x', view='layout',
                file_name='mmi_2x2_field', file_type='png')

        ## The bandwidth of the optimized 2x2 over 200 nm. The sections are built on
        ## the labeled profile 'pair' and on the default profile, so the wavelength
        ## is set on both before each solve.
        wavelengths = np.linspace(1450, 1650, 11)  # [nm]
        P_through, P_cross = [], []
        for wl in wavelengths:
            em.settings(wavelength=wl)
            em.settings(profile='pair', wavelength=wl)
            em.EME()
            S = em.get('S_matrix')
            P_through.append(np.abs(S.block('out.a', 'in.a')[0, 0]) ** 2)
            P_cross.append(np.abs(S.block('out.b', 'in.a')[0, 0]) ** 2)
        P_through, P_cross = np.array(P_through), np.array(P_cross)

        ## The band where the imbalance stays within 0.5 dB and the excess loss
        ## below 1 dB, read off the solved wavelengths by linear interpolation
        fine = np.linspace(wavelengths[0], wavelengths[-1], 1001)
        imbalance = np.interp(fine, wavelengths, 10 * np.log10(P_through / P_cross))
        excess_loss = np.interp(fine, wavelengths, -10 * np.log10(P_through + P_cross))
        band = fine[(np.abs(imbalance) <= 0.5) & (excess_loss <= 1)]
        print(f'2x2 band, imbalance within 0.5 dB and excess loss below 1 dB, from: {band.min():.0f} nm')
        print(f'2x2 band, imbalance within 0.5 dB and excess loss below 1 dB, to: {band.max():.0f} nm')
        em.close()

        plt.figure(figsize=(6, 3.5))
        plt.plot(wavelengths, 10 * np.log10(P_through), marker='o', ms=3, label='through')
        plt.plot(wavelengths, 10 * np.log10(P_cross), marker='o', ms=3, label='cross')
        plt.plot(wavelengths, 10 * np.log10(P_through + P_cross), color='gray', ls='--', label='total')
        plt.xlabel('Wavelength (nm)')
        plt.ylabel('Transmission (dB)')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.savefig('mmi_2x2_spectrum.png', dpi=300, bbox_inches='tight')
